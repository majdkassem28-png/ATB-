        function saveState() {

            localStorage.setItem('atb_cart', JSON.stringify(cart));

        // Language Switcher
        function toggleLanguage() {
            currentLang = currentLang === 'en' ? 'ar' : 'en';
            document.documentElement.lang = currentLang;
            document.documentElement.dir = currentLang === 'ar' ? 'rtl' : 'ltr';
            document.getElementById('langLabel').innerText = currentLang === 'en' ? 'العربية' : 'English';
            renderProducts();
            updateCartUI();

        // Render Store Products
        function renderProducts() {
            const grid = document.getElementById('productsGrid');
            const search = document.getElementById('searchInput').value.toLowerCase();
            grid.innerHTML = '';

            let filtered = products.filter(p => {
                let matchesBrand = currentBrandFilter === 'ALL' || p.brand === currentBrandFilter;
                let matchesSearch = p.title.toLowerCase().includes(search) || p.titleAr.toLowerCase().includes(search) || p.brand.toLowerCase().includes(search);
                return matchesBrand && matchesSearch;
            });

            if (filtered.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-gray-500">No products found matching your search.</div>`;
                return;
            }

            filtered.forEach((p, index) => {
                let realIndex = products.indexOf(p);
                let title = currentLang === 'ar' ? p.titleAr : p.title;
                let stockBadge = p.inStock 
                    ? `<span class="bg-green-100 text-green-800 text-xs font-bold px-2.5 py-1 rounded-full"><i class="fa-solid fa-check mr-1"></i> ${currentLang === 'ar' ? 'متوفر' : 'In Stock'}</span>`
                    : `<span class="bg-red-100 text-red-800 text-xs font-bold px-2.5 py-1 rounded-full"><i class="fa-solid fa-xmark mr-1"></i> ${currentLang === 'ar' ? 'غير متوفر' : 'Out of Stock'}</span>`;

                grid.innerHTML += `
                    <div class="bg-white rounded-xl shadow-sm hover:shadow-md transition border border-gray-200 overflow-hidden flex flex-col justify-between">
                        <div>
                            <div class="relative h-48 bg-gray-100 overflow-hidden">
                                <img src="${p.image}" alt="${title}" class="w-full h-full object-cover group-hover:scale-105 transition duration-300">
                                <div class="absolute top-3 left-3 flex gap-2">
                                    <span class="bg-brandNavy text-white text-xs font-black px-2.5 py-1 rounded shadow">${p.brand}</span>
                                </div>
                                <div class="absolute top-3 right-3">
                                    ${stockBadge}
                                </div>
                            </div>
                            <div class="p-5">
                                <span class="text-xs text-gray-500 font-medium uppercase tracking-wider">${p.category}</span>
                                <h3 class="font-bold text-gray-900 text-lg mt-1 mb-2 leading-snug">${title}</h3>
                                <div class="text-xl font-black text-brandRed mb-4">$${p.price.toFixed(2)}</div>
                            </div>
                        </div>
                        <div class="p-5 pt-0 flex flex-col gap-2">
                            ${p.inStock ? `
                                <button onclick="addToCart(${realIndex})" class="w-full bg-brandNavy hover:bg-blue-900 text-white font-bold py-2.5 rounded-lg text-sm shadow transition flex items-center justify-center gap-2">
                                    <i class="fa-solid fa-cart-plus"></i> ${currentLang === 'ar' ? 'أضف إلى السلة' : 'Add to Cart'}
                                </button>
                            ` : `
                                <button disabled class="w-full bg-gray-200 text-gray-400 font-bold py-2.5 rounded-lg text-sm cursor-not-allowed">
                                    ${currentLang === 'ar' ? 'نفذت الكمية' : 'Sold Out'}
                                </button>
                            `}
                            <a href="https://wa.me/96176542555?text=${encodeURIComponent('Hello ATB, I want to inquire about: ' + p.title + ' ($' + p.price + ')')}" target="_blank" class="w-full bg-green-600 hover:bg-green-700 text-white font-bold py-2 rounded-lg text-xs shadow transition flex items-center justify-center gap-1.5">
                                <i class="fa-brands fa-whatsapp text-sm"></i> ${currentLang === 'ar' ? 'اسأل على واتساب (+961 76 542 555)' : 'Inquire on WhatsApp'}
                            </a>
                        </div>
                    </div>
                `;
            });
        }

        function filterBrand(brand) {
            currentBrandFilter = brand;
            document.querySelectorAll('.brand-filter-btn').forEach(btn => {
                btn.classList.remove('bg-brandNavy', 'text-white');
                btn.classList.add('bg-gray-100', 'text-gray-700');
            });
            event.target.classList.remove('bg-gray-100', 'text-gray-700');
            event.target.classList.add('bg-brandNavy', 'text-white');
            renderProducts();
        }

            renderProducts();
        }

        // Cart Management
        function toggleCartDrawer() {
            const drawer = document.getElementById('cartDrawer');
            drawer.classList.toggle('hidden');
            renderCartItems();
        }

        function addToCart(index) {
            let prod = products[index];
            let existing = cart.find(item => item.index === index);
            if (existing) {
                existing.qty++;
            } else {
                cart.push({ index: index, product: prod, qty: 1 });
            }
            saveState();
            updateCartUI();
            toggleCartDrawer();
        }

        function updateCartUI() {
            let totalCount = cart.reduce((sum, item) => sum + item.qty, 0);
            document.getElementById('cartCountBadge').innerText = totalCount;
            document.getElementById('cartItemCount').innerText = totalCount;
        }

        function renderCartItems() {
            let list = document.getElementById('cartItemsList');
            list.innerHTML = '';
            if (cart.length === 0) {
                list.innerHTML = `<div class="text-center py-10 text-gray-400">Your cart is empty.</div>`;
                document.getElementById('cartSubtotal').innerText = '$0.00';
                document.getElementById('cartTotal').innerText = '$4.00';
                return;
            }

            let subtotal = 0;
            cart.forEach((item, i) => {
                let itemTotal = item.product.price * item.qty;
                subtotal += itemTotal;
                let title = currentLang === 'ar' ? item.product.titleAr : item.product.title;
                list.innerHTML += `
                    py-3 flex items-center justify-between gap-4">
                        <img src="${item.product.image}" class="w-16 h-16 object-cover rounded-lg border">
                        <div class="flex-1">
                            <h4 class="font-bold text-sm text-gray-900 line-clamp-1">${title}</h4>
                            <div class="text-xs text-brandRed font-bold">$${item.product.price.toFixed(2)}</div>
                            <div class="flex items-center gap-2 mt-1">
                                <button onclick="changeQty(${i}, -1)" class="w-6 h-6 bg-gray-200 rounded font-bold text-xs">-</button>
                                <span class="text-xs font-bold">${item.qty}</span>
                                <button onclick="changeQty(${i}, 1)" class="w-6 h-6 bg-gray-200 rounded font-bold text-xs">+</button>
                            </div>
                        </div>
                        <div class="text-right">
                            <div class="text-sm font-black text-gray-900">$${itemTotal.toFixed(2)}</div>
                            <button onclick="removeFromCart(${i})" class="text-red-500 hover:text-red-700 text-xs mt-1"><i class="fa-solid fa-trash"></i></button>
                        </div>
                    </div>
                `;
            });

            let deliveryFee = subtotal > 150 ? 0 : 4.00;
            let total = subtotal + deliveryFee;

            document.getElementById('cartSubtotal').innerText = '$' + subtotal.toFixed(2);
            document.getElementById('cartDeliveryFee').innerText = deliveryFee === 0 ? 'FREE ($0.00)' : '$4.00';
            document.getElementById('cartTotal').innerText = '$' + total.toFixed(2);
        }

        function changeQty(index, delta) {
            cart[index].qty += delta;
            if (cart[index].qty <= 0) {
                cart.splice(index, 1);
            }
            saveState();
            renderCartItems();
            updateCartUI();
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            saveState();
            renderCartItems();
            updateCartUI();
        }

        // Checkout & WhatsApp Order Submission
        function openCheckoutModal() {
            if (cart.length === 0) {
                alert("Your cart is empty!");
                return;
            }
            toggleCartDrawer();
            document.getElementById('checkoutModal').classList.remove('hidden');
        }

        function closeCheckoutModal() {
            document.getElementById('checkoutModal').classList.add('hidden');
        }

        function submitOrder(e) {
            e.preventDefault();
            let name = document.getElementById('custName').value;
            let phone = document.getElementById('custPhone').value;
            let city = document.getElementById('custCity').value;
            let address = document.getElementById('custAddress').value;
            let notes = document.getElementById('custNotes').value;

            let subtotal = cart.reduce((sum, item) => sum + (item.product.price * item.qty), 0);
            let delivery = subtotal > 150 ? 0 : 4.00;
            let total = subtotal + delivery;

            let orderId = '#ATB-' + Math.floor(1000 + Math.random() * 9000);
            let newOrder = {
                id: orderId,
                date: new Date().toLocaleString(),
                name: name,
                phone: phone,
                city: city,
                address: address,
                notes: notes,
                items: [...cart],
                subtotal: subtotal,
                total: total,
                status: 'Pending'
            };

            orders.unshift(newOrder);
            saveState();

            // Construct WhatsApp message
            let waMsg = `🛒 *New Order from ATB Website* (${orderId})\n\n`;
            waMsg += `👤 *Customer:* ${name}\n`;
            waMsg += `📞 *Phone:* ${phone}\n`;
            waMsg += `📍 *City:* ${city} - ${address}\n`;
            if(notes) waMsg += `📝 *Notes:* ${notes}\n\n`;
            waMsg += `📦 *Items Ordered:*\n`;
            cart.forEach(item => {
                waMsg += `- ${item.product.title} (x${item.qty}) : $${(item.product.price * item.qty).toFixed(2)}\n`;
            });
            waMsg += `\n💰 *Total Amount:* $${total.toFixed(2)} (Cash on Delivery)\n`;

            cart = [];
            saveState();
            updateCartUI();
            closeCheckoutModal();

            // Open WhatsApp to +961 76 542 555
            window.open(`https://wa.me/96176542555?text=${encodeURIComponent(waMsg)}`, '_blank');
            alert('Order placed successfully! Order ID: ' + orderId);
        }

        function renderSellerOrders() {
            document.getElementById('orderCountBadge').innerText = orders.length;
            let list = document.getElementById('sellerOrdersList');
            list.innerHTML = '';
            if (orders.length === 0) {
                list.innerHTML = `<div class="text-center py-12 text-gray-400 bg-gray-50 rounded-xl">No customer orders received yet.</div>`;

                        <div>
                            <div class="flex items-center gap-2 mb-1">
                                <span class="font-bold text-brandRed">${o.id}</span>
                                <span class="text-xs text-gray-400">${o.date}</span>
                                <span class="px-2 py-0.5 rounded-full text-xs font-bold ${statusColor}">${o.status}</span>

                            <div class="text-xs font-extrabold text-gray-900 mt-1">Total: $${o.total.toFixed(2)}</div>

                            <select onchange="updateOrderStatus(${i}, this.value)" class="text-xs border rounded p-1.5 font-bold">
                                <option value="Pending" ${o.status === 'Pending' ? 'selected' : ''}>Pending</option>
                                <option value="Out for Delivery" ${o.status === 'Out for Delivery' ? 'selected' : ''}>Out for Delivery</option>
                                <option value="Delivered" ${o.status === 'Delivered' ? 'selected' : ''}>Delivered</option>
                            </select>
                            <a href="https://wa.me/961${o.phone.replace(/^0+/, '')}?text=${encodeURIComponent('Hello ' + o.name + ', regarding your order ' + o.id + ' at ATB:')}" target="_blank" class="bg-green-600 text-white px-3 py-1.5 rounded text-xs font-bold flex items-center gap-1">

            orders[index].status = status;
            saveState();
            renderSellerOrders();
        }

        function deleteOrder(index) {
            if (confirm('Delete this order record?')) {
                orders.splice(index, 1);
                saveState();
                renderSellerOrders();
            }
        }

        function saveProduct(e) {
            e.preventDefault();
            let editIdx = parseInt(document.getElementById('editIndex').value);
            let newProd = {
                title: document.getElementById('prodTitle').value,
                titleAr: document.getElementById('prodTitleAr').value,
                brand: document.getElementById('prodBrand').value,
                category: document.getElementById('prodCategory').value,
                price: parseFloat(document.getElementById('prodPrice').value),
                inStock: document.getElementById('prodInStock').value === 'true',
                image: document.getElementById('prodImage').value
            };

            if (editIdx >= 0) {
                products[editIdx] = newProd;
            } else {
                products.unshift(newProd);
            }

            saveState();
            resetProductForm();
            renderProducts();
            renderSellerProductsTable();
        }

        function editProduct(index) {
            let p = products[index];
            document.getElementById('editIndex').value = index;
            document.getElementById('prodTitle').value = p.title;
            document.getElementById('prodTitleAr').value = p.titleAr;
            document.getElementById('prodBrand').value = p.brand;
            document.getElementById('prodCategory').value = p.category;
            document.getElementById('prodPrice').value = p.price;
            document.getElementById('prodInStock').value = p.inStock ? 'true' : 'false';
            document.getElementById('prodImage').value = p.image;
            document.getElementById('formTitle').innerText = 'Edit Product #' + (index + 1);
        }

        function deleteProduct(index) {
            if (confirm('Are you sure you want to delete this product?')) {
                products.splice(index, 1);
                saveState();
                renderProducts();
                renderSellerProductsTable();
            }
        }

        function resetProductForm() {
            document.getElementById('editIndex').value = -1;
            document.getElementById('productForm').reset();
            document.getElementById('formTitle').innerText = 'Add New Product';
        }

        window.onload = function() {
            renderProducts();
            updateCartUI();
        };
